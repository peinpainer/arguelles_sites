C++ Activities

Activity 1: Count Positive and Negative Numbers

#include <iostream>
using namespace std;

int main() {
    int num;
    int positive = 0, negative = 0;
    do {
        cout << "Enter a number: ";
        cin >> num;

        if (num > 0) positive++;
        else if (num < 0) negative++;

    } while (num != 0);

    cout << "You entered " << positive << " positive numbers and "
         << negative << " negative numbers." << endl;
    return 0;
}

Activity 2: Factorial of a Number

#include <iostream>
using namespace std;

int main() {
    int num;
    int factorial = 1;
    int i = 1;

    cout << "Enter a number: ";
    cin >> num;

    while (i <= num) {
        factorial *= i;
        i++;
    }

    cout << "Factorial = " << factorial << endl;
    return 0;
}

Activity 3: Fibonacci and Sum

#include <iostream>
using namespace std;

int main() {
    char choice;
    do {
        int n;
        cout << "Enter number of terms: ";
        cin >> n;

        int a = 0, b = 1, next, sum = 0;

        cout << "Fibonacci: ";
        for (int i = 1; i <= n; i++) {
            cout << a << " ";
            sum += a;
            next = a + b;
            a = b;
            b = next;
        }

        cout << " Sum = " << sum << endl;

        cout << "Do you want to run again? (y/n): ";
        cin >> choice;

    } while (choice == 'y' || choice == 'Y');

    return 0;
}

Activity 2: Static Counter Function

#include <stdio.h>

void counter() {
    static int count = 0;
    count++;
    printf("Function called %d times\n", count);
}

int main() {
    counter();
    counter();
    counter();
    return 0;
}

Activity 2: Sum of Two Numbers

import java.util.Scanner;
public class SumNumbers {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("Enter first number: ");
        int num1 = scanner.nextInt();
        System.out.print("Enter second number: ");
        int num2 = scanner.nextInt();
        int sum = num1 + num2;
        System.out.println("The sum is: " + sum);
        scanner.close();
    }
}

Activity 3: Even or Odd

import java.util.Scanner;
public class EvenOdd {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("Enter a number: ");
        int number = scanner.nextInt();
        if (number % 2 == 0) {
            System.out.println(number + " is even.");
        } else {
            System.out.println(number + " is odd.");
        }
        scanner.close();
    }
}
